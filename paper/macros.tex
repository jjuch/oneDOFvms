\def\SHOWGAPS{1}
\newcounter{gapitem}
\newcommand{\gaplist}{}
\makeatletter
\newcommand{\gap}[2]{%
  \refstepcounter{gapitem}%
  \edef\@gapentry{\noexpand\item[G\thegapitem]\noexpand\textbf{\unexpanded{#1}}
    (p.\noexpand~\thepage)}%
  \expandafter\g@addto@macro\expandafter\gaplist\expandafter{\@gapentry}%
  {\color{BrickRed}\textbf{[Gap~G\thegapitem: #1]}\ \textit{#2}}}
